  <h2>📊 Canlı Panel - {{SHEET_NAME}}</h2>
  <div id="table-container">Veriler yükleniyor...</div>

  <button onclick="exportToExcel()">Excel'e Aktar</button>
  <script>
    const url = "https://script.google.com/macros/s/{{WEBAPP_ID}}/exec";
    fetchData(); setInterval(fetchData, 30000);
    function fetchData() {
      fetch(url).then(r => r.json()).then(d => {
        const h = d[0], r = d.slice(1);
        let html = "<table><thead><tr>" + h.map(c => "<th>" + c + "</th>").join("") + "</tr></thead><tbody>";
        r.forEach(row => { html += "<tr>" + row.map(c => "<td>" + c + "</td>").join("") + "</tr>"; });
        html += "</tbody><tfoot><tr>" + h.map((c, i) => i === 0 ? "<td>Toplam</td>" : "<td>" + r.reduce((s, v) => s + (+v[i] || 0), 0).toFixed(2) + "</td>").join("") + "</tr>";
        html += "<tr>" + h.map((c, i) => { if (i === 0) return "<td>Oran</td>"; let base = r.reduce((s, v) => s + (+v[1] || 0), 0); let total = r.reduce((s, v) => s + (+v[i] || 0), 0); return "<td>" + (base ? (total / base * 100).toFixed(2) + "%" : "-") + "</td>"; }).join("") + "</tr></tfoot></table>";
        document.getElementById("table-container").innerHTML = html;

        const labels = r.map(x => x[0]);
        const values = r.map(x => parseFloat(x[1]) || 0);
        new Chart(document.getElementById("chart"), {
          type: "pie",
          data: { labels, datasets: [{ label: h[1], data: values }] },
          options: { plugins: { title: { display: true, text: h[1] } } }
        });
      });
    }

    function exportToExcel() {
      const csv = [...document.querySelectorAll("table tr")].map(row => [...row.cells].map(c => c.innerText).join(";")).join("\n");
      const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
      const a = document.createElement("a");
      a.href = URL.createObjectURL(blob);
      a.download = "panel.csv";
      a.click();
    }
  </script>
