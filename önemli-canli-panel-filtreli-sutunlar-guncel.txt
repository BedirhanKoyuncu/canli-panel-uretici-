  <script>
    const sheetName = "Sheet1";
    const webAppUrl = "https://script.google.com/macros/s/AKfycbwmImZN_bXWMKMg0utg0gnv-FsV9KrNbhIrP8QrYL7gQEtS34FG1uz8_rTZb3RkgKNB6Q/exec";
    let currentData = [];
    let chart;

    function fetchData() {
      fetch(`${webAppUrl}?sheet=${sheetName}`)
        .then(response => response.json())
        .then(data => {
          currentData = data;
          renderTable(data);
          renderChart("bar");
        })
        .catch(err => {
          document.getElementById("table-container").innerText = "Veri alınamadı: " + err.message;
        });
    }

    function renderTable(data) {
  let html = "<table><thead><tr>";
  data[0].forEach(col => html += `<th>${col}</th>`);
  html += "<th>Hedef Toplamı</th><th>Gerçek Toplamı</th><th>Oran</th></tr></thead><tbody>";

  const labelRow = data[1];  // 2. satırda hedef/gerçek ayrımı var
  const hedefIndices = [], gercekIndices = [];

  labelRow.forEach((val, i) => {
    const lower = val.toLowerCase();
    if (lower.includes("hedef")) hedefIndices.push(i);
    if (lower.includes("gerçek")) gercekIndices.push(i);
  });

  let totalHedef = 0, totalGercek = 0;

  for (let i = 2; i < data.length; i++) {
    let rowHedef = 0, rowGercek = 0;
    html += "<tr>";
    data[i].forEach((cell, j) => {
      html += `<td>${cell}</td>`;
      const val = parseFloat(cell.toString().replace(",", "."));
      if (!isNaN(val)) {
        if (hedefIndices.includes(j)) rowHedef += val;
        if (gercekIndices.includes(j)) rowGercek += val;
      }
    });
    totalHedef += rowHedef;
    totalGercek += rowGercek;

    html += `<td>${rowHedef.toFixed(2)}</td>`;
    html += `<td>${rowGercek.toFixed(2)}</td>`;
    const oran = rowHedef > 0 ? (rowGercek / rowHedef) * 100 : 0;
    html += `<td>${oran.toFixed(1)}%</td></tr>`;
  }

  html += "</tbody><tfoot><tr>";
  for (let j = 0; j < data[0].length; j++) {
    html += j === 0 ? "<td><b>Genel Toplam</b></td>" : "<td></td>";
  }
  html += `<td><b>${totalHedef.toFixed(2)}</b></td>`;
  html += `<td><b>${totalGercek.toFixed(2)}</b></td>`;
  const genelOran = totalHedef > 0 ? (totalGercek / totalHedef) * 100 : 0;
  html += `<td><b>${genelOran.toFixed(1)}%</b></td></tr>`;

  // Yeni satır: Sütunların altına toplamları yaz
  html += "<tr>";
  const colSums = new Array(data[0].length).fill(0);
  for (let i = 2; i < data.length; i++) {
    data[i].forEach((cell, j) => {
      const val = parseFloat(cell.toString().replace(",", "."));
      if (!isNaN(val)) colSums[j] += val;
    });
  }
  colSums.forEach((sum, j) => {
    html += j === 0 ? "<td><b>Sütun Toplamı</b></td>" : `<td><b>${sum.toFixed(2)}</b></td>`;
  });
  html += "<td></td><td></td><td></td></tr></tfoot></table>";

  document.getElementById("table-container").innerHTML = html;
}

    function renderChart(type) {
      if (!currentData.length) return;

      const labels = currentData.slice(1).map(row => row[0]);
      const values = currentData.slice(1).map(row => parseFloat(row[1].toString().replace(",", ".")) || 0);
      const ctx = document.getElementById("myChart").getContext("2d");
      if (chart) chart.destroy();
      chart = new Chart(ctx, {
        type,
        data: {
          labels,
          datasets: [{
            label: currentData[0][1],
            data: values,
            backgroundColor: ['#007BFF','#28A745','#FFC107','#DC3545','#6610f2','#20c997'],
            borderColor: '#000',
            borderWidth: 1
          }]
        },
        options: {
          responsive: true,
          plugins: {
            legend: { display: type !== 'bar' && type !== 'line' },
            title: { display: true, text: `Grafik Türü: ${type}` }
          }
        }
      });
    }

function exportCSV() {
  const table = document.querySelector("table");
  if (!table) return;
  let csv = "";
  table.querySelectorAll("tr").forEach(row => {
    const rowData = Array.from(row.querySelectorAll("th,td")).map(cell =>
      `"${cell.textContent.replace(/"/g, '""')}"`
    ).join(";");
    csv += rowData + "\n";
  });
  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
  const link = document.createElement("a");
  link.href = URL.createObjectURL(blob);
  link.download = "panel-veri.csv";
  link.click();
}

async function exportPDF() {
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF();
  const table = document.querySelector("table");
  if (!table) return;
  let y = 10;
  table.querySelectorAll("tr").forEach(row => {
    let x = 10;
    row.querySelectorAll("th,td").forEach(cell => {
      doc.text(cell.textContent, x, y);
      x += 40;
    });
    y += 10;
  });
  doc.save("panel-veri.pdf");
}
